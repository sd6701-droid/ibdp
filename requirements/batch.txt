                          # Qwen2.5-VL natively (that landed in 4.49), so the
                          # 72B needs NO extra architecture package.
accelerate>=1.0           # required for device_map="auto" -- i.e. for sharding

qwen-vl-utils==0.0.14

# --- Qwen3-Omni (native audio+video) ---------------------------------------
# A DIFFERENT package from qwen-vl-utils, not a newer one -- both are installed
# and both are used. process_mm_info() takes use_audio_in_video and returns an
# audio list qwen_vl_utils has no concept of.
#
# NB: qwen_omni_utils takes a whole file, NOT a [start,end] range, which is why
# the omni backend re-cuts each segment to a temp mp4 instead of decoding a
# sub-range in place like every other backend.
qwen-omni-utils
soundfile                 # qwen_omni_utils reads audio through it; the import
                          # is unguarded, so its absence surfaces as an
                          # ImportError from inside the util, not a clear error.
librosa                   # resampling to the 16k the audio tower expects

# --- InternVL3 (OpenGVLab/InternVL3-*) -------------------------------------
# Its architecture is NOT in transformers. The modelling code ships inside the
# checkpoint (modeling_internvl_chat.py, modeling_intern_vit.py, conversation.py)
# and loads via trust_remote_code=True -- so nothing to install for the arch
# itself. But that vendored code hard-imports the two packages below, with no
# try/except, and the failure surfaces as an ImportError from deep inside HF's
# dynamic module loader where it looks like a transformers bug.
#
# flash_attn and apex are ALSO imported there, but guarded -- InternVL sets
# use_flash_attn=False when they are absent. On A100 that costs some speed and
# nothing else, so neither is listed here (flash-attn is a painful source build).
einops                    # modeling_intern_vit.py: `from einops import rearrange`
timm>=0.9                 # modeling_intern_vit.py: `from timm.layers import DropPath`
                          # timm.layers is 0.9+; on 0.6 it is timm.models.layers.
torchcodec                # video decode backend. NOT decord -- Qwen recommends
                          # against it (decode hangs, unmaintained).
pymupdf>=1.24             # PDF -> PNG. NOT pdf2image: that shells out to a
                          # Poppler binary you cannot install on an HPC node.
wandb                     # optional: scripts/26 --wandb. Captures GPU util,
                          # VRAM, power and temp automatically via NVML. Use
                          # WANDB_MODE=offline on compute nodes -- online mode
                          # blocks on init there and then drops metrics.
pydantic>=2
openai>=1.40
pyyaml>=6
tqdm
